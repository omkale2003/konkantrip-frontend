function getStatusClasses(status) {
  switch (status) {
    case "Approved":
      return "bg-emerald-50 text-emerald-700 border-emerald-200";
 
    case "Pending":
    case "Under Review":
      return "bg-amber-50 text-amber-700 border-amber-200";
 
    case "Rejected":
    case "Suspended":
    case "Inactive":
      return "bg-red-50 text-red-700 border-red-200";
 
    case "Draft":
    default:
      return "bg-slate-100 text-slate-600 border-slate-200";
  }
}
 
function getStatusIcon(status) {
  switch (status) {
    case "Approved":
      return CheckCircle2;
 
    case "Pending":
    case "Under Review":
      return Clock3;
 
    case "Rejected":
    case "Suspended":
    case "Inactive":
      return XCircle;
 
    default:
      return Building2;
  }
}

  const firstName = owner?.first_name || "Owner";
 
  const {

    page: 1,
    limit: 10,
    owner_id: owner?.p_owner_id,

  const properties = useMemo(() => (Array.isArray(data?.data) ? data.data : []), [data]);
  const totalProperties = data?.pagination?.total ?? properties.length;

    () => properties.filter((p) => p.property_status === "Approved").length,
    [properties]

    () => properties.reduce((acc, p) => acc + (Number(p.total_rooms) || 0), 0),
    [properties]

                    >

                    </Link>
                  </div>
                );
              })}
            </div>
          )}
        </section>
      )}
    </div>
  );
}
